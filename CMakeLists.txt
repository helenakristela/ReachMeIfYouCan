cmake_minimum_required(VERSION 3.10)
project(ReachMeIfYouCan CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(raylib REQUIRED)
add_executable(reach_me src/main.cpp)
target_include_directories(reach_me PRIVATE external)
target_link_libraries(reach_me raylib)
if (APPLE)
target_link_libraries(reach_me "-framework IOKit" "-framework Cocoa" "-framework OpenGL")
endif()
if (UNIX AND NOT APPLE)
target_link_libraries(reach_me m pthread dl rt X11)
endif()
